import java.sql.Connection;

public class StudentManager {

    private static Scanner scan = new Scanner(System.in);

    public static void main(String[] args) {
        Connection conn = null;
        Statement stmt = null;
        try {
            conn = DBUtil.getConnection();
            stmt = conn.createStatement();
            boolean isRun = true;

            while (isRun) {
                showMenu();
                System.out.print("请输入功能序号：");
                int choice = scan.nextInt();
                scan.nextLine();

                switch (choice) {
                    case 1: addClass(stmt); break;
                    case 2: addStudent(stmt); break;
                    case 3: queryStudentByClass(stmt); break;
                    case 4: countStudentByClass(stmt); break;
                    case 5: changeClass(stmt); break;
                    case 6: stopStudy(stmt); break;
                    case 7: updateStudent(stmt); break;
                    case 8: reStudy(stmt); break;
                    case 9: showStudentChange(stmt); break;
                    case 10: findMostChangeStudent(stmt); break;
                    case 11: updateMajor(stmt); break;
                    case 12: studentSelfChange(stmt); break;
                    case 13: studentSelfScore(stmt); break;
                    case 14: sumScore(stmt); break;
                    case 15: avgScore(stmt); break;
                    case 16: sumPoint(stmt); break;
                    case 17: isRun = false; System.out.println("谢谢使用！"); break;
                    default: System.out.println("输入错误！");
                }
            }
        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            DBUtil.close(conn, stmt);
        }
    }

    private static void addClass(Statement stmt) {

    private static void queryStudentByClass(Statement stmt) {
        try {

    }

    private static void countStudentByClass(Statement stmt) {
        try {
            String sql = "SELECT c.cid,c.cname,COUNT(s.sid) num FROM class c LEFT JOIN student s ON c.cid=s.cid GROUP BY c.cid";

            System.out.println("班级编号\t班级名称\t人数");
            while (rs.next()) {
                System.out.println(rs.getInt("cid") + "\t\t" + rs.getString("cname") + "\t\t" + rs.getInt("num"));
            }
            DBUtil.close(null, null, rs);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    private static void changeClass(Statement stmt) {
        try {
            System.out.println("\n===== 学生列表 =====");
            String sqlShow = "SELECT sid, sname, cid, zy, zt FROM student";
            ResultSet rsShow = stmt.executeQuery(sqlShow);
            System.out.println("学号\t姓名\t班级ID\t专业\t状态");
            while (rsShow.next()) {
                System.out.println(rsShow.getInt("sid") + "\t" + rsShow.getString("sname") + "\t"
                        + rsShow.getInt("cid") + "\t" + rsShow.getString("zy") + "\t" + rsShow.getInt("zt"));
            }
            DBUtil.close(null, null, rsShow);

            System.out.print("\n请输入降班学生编号：");
            int sid = scan.nextInt();

            String sqlOld = "SELECT s.cid cid, s.sname sname, s.zy zy, s.zt zt, SUBSTRING_INDEX(c.cname,'年',1)+0 grade "
                    + "FROM student s JOIN class c ON s.cid=c.cid WHERE sid=" + sid;
            ResultSet rsOld = stmt.executeQuery(sqlOld);
            if (!rsOld.next()) {
                System.out.println("学生不存在");
                DBUtil.close(null, null, rsOld);
                return;
            }
            int oldCid = rsOld.getInt("cid");
            String oldSname = rsOld.getString("sname");
            String oldZy = rsOld.getString("zy");
            int oldZt = rsOld.getInt("zt");
            int nowGrade = rsOld.getInt("grade");
            DBUtil.close(null, null, rsOld);

            String sqlCan = "SELECT cid, cname, SUBSTRING_INDEX(cname,'年',1)+0 grade FROM class "
                    + "WHERE SUBSTRING_INDEX(cname,'年',1)+0 < " + nowGrade;
            ResultSet rsCan = stmt.executeQuery(sqlCan);
            System.out.println("\n可降班级：");
            System.out.println("编号\t名称\t年级");
            boolean has = false;
            while (rsCan.next()) {
                has = true;
                System.out.println(rsCan.getInt("cid") + "\t" + rsCan.getString("cname") + "\t" + rsCan.getInt("grade"));
            }
            DBUtil.close(null, null, rsCan);
            if (!has) {
                System.out.println("已是最低年级");
                return;
            }

            System.out.print("请输入新班级编号：");
            int newCid = scan.nextInt();

            String sqlUpdate = "UPDATE student SET cid=" + newCid + " WHERE sid=" + sid;
            stmt.executeUpdate(sqlUpdate);

            String sqlLog = "INSERT INTO student_change(sid,old_cid,new_cid,old_sname,new_sname,old_zy,new_zy,old_zt,new_zt) "
                    + "VALUES(" + sid + "," + oldCid + "," + newCid + ",'" + oldSname + "','" + oldSname + "','" + oldZy + "','" + oldZy + "'," + oldZt + "," + oldZt + ")";
            stmt.executeUpdate(sqlLog);

            System.out.println("降班成功，变动已记录");

        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    private static void findMostChangeStudent(Statement stmt) {

            if (rs.next()) {
                System.out.println("变动最多的学生学号：" + rs.getInt("sid") + "，次数：" + rs.getInt("cnt"));
            }

            e.printStackTrace();
        }
    }

    private static void sumScore(Statement stmt) {
        try {

    private static void avgScore(Statement stmt) {
        try {
            System.out.print("学号：");
            int sid = scan.nextInt();
            String sql = "SELECT AVG(cj) avg FROM score WHERE sid=" + sid;
            ResultSet rs = stmt.executeQuery(sql);
            if (rs.next()) System.out.println("平均成绩：" + String.format("%.2f", rs.getDouble("avg")));
            DBUtil.close(null, null, rs);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
