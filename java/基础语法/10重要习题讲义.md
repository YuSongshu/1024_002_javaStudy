  1.普通连接  WHERE 一方的主键=多方的外键

  4.外连接   OUTER JOIN   ON 一方的主键=多方的外键

CREATE TABLE qqnum(id INT(2) PRIMARY KEY AUTO_INCREMENT,num VARCHAR(10));

CREATE TABLE qqrole(id INT(2) PRIMARY KEY AUTO_INCREMENT,rname VARCHAR(10));

CREATE TABLE num_role(n_id INT(2),r_id INT(2));



关系列 n.id = nr.n_id AND nr.r_id = r.id

SELECT *  FROM qqnum n,qqrole r,num_role nr

WHERE n.id = nr.n_id AND nr.r_id = r.id
